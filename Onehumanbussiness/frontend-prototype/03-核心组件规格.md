```typescript
// 使用 React Flow 实现

interface EmployeeOrgChartProps {
  employees: AIEmployee[];

interface OrgChartNode {
  id: string;                     // employee codename
  type: 'employee' | 'group';
  position: { x: number; y: number };
  data: {
    name: string;
    avatar: string;
    level: number;
    status: EmployeeStatus;
    currentTask?: string;
    role: string;
  };

interface OrgChartEdge {
  id: string;
  source: string;                 // 汇报关系
  target: string;
  type: 'solid' | 'dashed';       // 实线=直接汇报，虚线=协作关系
}

// 交互：
// - 点击节点 → 打开EmployeeDetail
// - 拖拽节点 → 调整组织架构布局（仅视觉，不改变汇报关系）
// - 双击节点 → 快速派发任务
// - 右键菜单 → 查看档案 / 培训 / 解雇
```

### 4.4 TaskReviewForm

```typescript
interface TaskReviewFormProps {
  onSubmit: (review: TaskReview) => void;
  defaultValues?: Partial<TaskReview>;

interface TaskReview {
  overallRating: number;          // 1-5
  dimensionRatings: {
    quality: number;
    timeliness: number;
    comprehension: number;
    innovation: number;
    communication: number;
  };
  whatWorked: string;             // 至少20字

// - 文字反馈必填且最少20字
// - 评分必填
// - 选择reject时必须填写原因
```

## 六、状态管理（Zustand Store）

### 6.1 OHB Store 划分

```typescript
// stores/ohbStore.ts

interface OHBState {
  // 当前公司
  company: OneCompany | null;
  setCompany: (company: OneCompany) => void;
  
  // AI员工
  employees: AIEmployee[];
  fetchEmployees: () => Promise<void>;
  hireEmployee: (roleType: string) => Promise<void>;
  dismissEmployee: (employeeId: string) => Promise<void>;
  
  // 任务
  tasks: Task[];
  currentTask: Task | null;
  setCurrentTask: (task: Task | null) => void;
  createTask: (task: TaskCreateRequest) => Promise<void>;
  reviewTask: (taskId: string, review: TaskReview) => Promise<void>;
  
  // 流水线
  currentStage: VentureStage;
  sprints: Sprint[];
  milestones: Milestone[];
  
  // 实时状态
  wsConnected: boolean;
  notifications: Notification[];
}

// 使用 Zustand 的 persist 中间件持久化到 localStorage
// 使用 subscribe 监听状态变化，自动同步到后端
```

### 6.2 WebSocket 事件处理

```typescript
// stores/wsStore.ts

interface WSEvents {
```
