## 二、执行流程

### 2.1 主流程

```python
async def execute_task(task_id: str) -> Deliverable:
    task = await load_task(task_id)
    employee = await load_employee(task.assignee_id)
    
    # 1. 需求解析
    parsed_requirement = await parse_requirement(task)
    
    # 2. 生成执行计划
    execution_plan = await generate_execution_plan(
        task, employee, parsed_requirement
    )
    
    # 3. 执行计划
    results = []
    for step in execution_plan.steps:
        # 更新进度
        await update_task_progress(task_id, step)
        
        # 执行步骤
        step_result = await execute_step(step, employee, task)
        results.append(step_result)
        
        # 检查是否需要暂停
        if step_result.requires_attention:
            await pause_task(task_id, step_result.attention_request)
            return  # 等待学生响应后继续
        
        # 检查是否出错
        if step_result.status == 'error':
            recovery = await attempt_recovery(step, step_result.error)
            if not recovery.success:
                await fail_task(task_id, step_result.error)
                return
    
    # 4. 整合结果
    deliverable = await compose_deliverable(task, results)
    
    # 5. 伦理审计
    audit_result = await ethos_audit(deliverable)
    if audit_result.severity == 'BLOCK':
        await block_task(task_id, audit_result.reason)
        return
    
    # 6. 提交完成
    await complete_task(task_id, deliverable)
```

### 2.2 需求解析器

```python
class RequirementParser:
    """将学生的自然语言需求解析为结构化任务"""
    
    async def parse(self, task: AITask) -> ParsedRequirement:
        # 1. 提取核心目标
        prompt = f"""
        分析以下任务需求，提取结构化信息：
        
        任务标题: {task.title}
        任务描述: {task.description}
        验收标准: {json.dumps(task.acceptance_criteria)}
        
        请输出JSON格式：
        {{
          "core_goal": "核心目标（一句话）",
          "sub_tasks": ["子任务1", "子任务2"],
          "required_skills": ["需要的技能"],
          "required_tools": ["需要的工具"],
          "constraints": ["约束条件"],
          "ambiguities": ["模糊点，需要澄清"],
          "success_criteria": ["成功标准"]
        }}
        """
        
        result = await llm_generate(prompt, model="gpt-4o-mini")
        parsed = json.loads(result)
        
        # 2. 检查模糊点
        if parsed['ambiguities']:
            await request_clarification(task.id, parsed['ambiguities'])
            raise ClarificationNeeded()
        
        return ParsedRequirement(**parsed)
```

### 2.3 执行计划生成

```python
class ExecutionPlanner:
    """根据任务类型生成执行步骤"""
    
    async def generate_plan(
        self, 
        task: AITask, 
        employee: AIEmployee,
        requirement: ParsedRequirement
    ) -> ExecutionPlan:
        
        # 根据任务类型选择计划模板
        templates = {
            'code': CodeExecutionTemplate(),
            'design': DesignExecutionTemplate(),
            'copywriting': CopyExecutionTemplate(),
            'research': ResearchExecutionTemplate(),
            'analysis': AnalysisExecutionTemplate(),
        }
        
        template = templates.get(task.task_type, GenericExecutionTemplate())
        
        # 生成步骤
        steps = template.generate_steps(requirement, employee)
        
        # 估算时间
        estimated_duration = sum(s.estimated_time for s in steps)
        
        return ExecutionPlan(
            steps=steps,
            estimated_duration=estimated_duration,
            required_tools=list(set(
                tool for step in steps for tool in step.required_tools
            ))
        )
```

async def execute_step(
    step: ExecutionStep,
    employee: AIEmployee,
    task: AITask
) -> StepResult:
    
    # 1. 准备上下文
    context = await build_context(employee, task, step)
    
    # 2. 构建Prompt
    prompt = build_step_prompt(step, context, employee.personality_prompt)
    
    # 3. 调用LLM
    llm_response = await llm_generate(
        prompt,
        model=select_model(step.complexity),
        temperature=0.7,
        max_tokens=estimate_tokens(step)
    )
    
    # 4. 解析输出
    parsed_output = parse_llm_output(llm_response, step.output_format)
    
    # 5. 如果需要工具调用
    if step.requires_tools:
        tool_results = await execute_tools(
            step.required_tools, 
            parsed_output.tool_calls,
            task.company_id
        )
        parsed_output.tool_results = tool_results
    
    # 6. 后处理
    final_output = await post_process(parsed_output, step)
    
    return StepResult(
        status='success',
        output=final_output,
        duration=time.time() - start_time
    )
```

def select_model(complexity: str, task_type: str) -> str:
    """根据任务复杂度选择最合适的模型"""
    
    routing_table = {
        'simple': {
            'default': 'gpt-4o-mini',      # 便宜、快速
            'code': 'claude-3-haiku',       # 代码能力好
        },
        'normal': {
            'default': 'gpt-4o',
            'code': 'claude-3-sonnet',
            'creative': 'claude-3-sonnet',
        },
        'complex': {
            'default': 'claude-3-opus',     # 推理能力强
            'code': 'claude-3-opus',
            'strategy': 'gpt-4o',
        }
    }
    
    return routing_table.get(complexity, {}).get(task_type, 
           routing_table.get(complexity, {}).get('default', 'gpt-4o'))
```

class CostController:

### 4.1 工具选择逻辑

class ToolSelector:
    """为每个执行步骤选择最合适的工具组合"""
    
    async def select_tools(
        self, 
        step: ExecutionStep,
        available_tools: List[MCPTool]
    ) -> List[MCPTool]:
        
        # 1. 基于步骤类型的默认工具
        default_tools = self.get_default_tools(step.type)
        
        # 2. 基于需求描述的动态选择
        prompt = f"""
        给定以下执行步骤，从可用工具中选择最合适的：
        
        步骤: {step.description}
        可用工具: {[t.name for t in available_tools]}
        
        请输出需要调用的工具列表和参数。
        """
        
        selection = await llm_generate(prompt, model="gpt-4o-mini")
        selected = parse_tool_selection(selection)
        
        # 3. 权限检查
        authorized = [
            t for t in selected 
            if t.name in employee.authorized_tools
        ]
        
        return authorized
```

async def execute_tools(
    tools: List[MCPTool],
    tool_calls: List[ToolCall],
    company_id: str
) -> List[ToolResult]:
    
    results = []
    
    for call in tool_calls:
        tool = find_tool(tools, call.tool_name)
        
        # 1. 参数校验
        validated_params = validate_params(call.params, tool.input_schema)
        
        # 2. 执行调用
        try:
            result = await tool.execute(validated_params)
            results.append(ToolResult(
                tool_name=call.tool_name,
                status='success',
                data=result
            ))
        except Exception as e:
            results.append(ToolResult(
                tool_name=call.tool_name,
                status='error',
                error=str(e)
            ))
            
            # 记录失败
            await log_mcp_call(company_id, call, error=str(e))
    
    return results
```

## 五、错误恢复机制

### 5.1 分级恢复策略

```python
async def attempt_recovery(step: ExecutionStep, error: Exception) -> RecoveryResult:
    """
    分级错误恢复：
    Level 1: 重试（网络超时、临时错误）
    Level 2: 参数调整（工具调用失败）
    Level 3: 降级策略（换模型、简化需求）
    Level 4: 人工介入（暂停任务）
    """
    
    # Level 1: 重试
    if is_retryable(error):
        for attempt in range(3):
            await asyncio.sleep(2 ** attempt)  # 指数退避
            try:
                result = await execute_step(step)
                return RecoveryResult(success=True, method='retry')
            except:
                continue
    
    # Level 2: 调整参数
    if isinstance(error, ToolExecutionError):
        adjusted_step = adjust_tool_params(step, error)
        try:
            result = await execute_step(adjusted_step)
            return RecoveryResult(success=True, method='param_adjustment')
        except:
            pass
    
    # Level 3: 降级
    if step.complexity == 'complex':
        simplified_step = simplify_step(step)
        try:
            result = await execute_step(simplified_step)
            return RecoveryResult(
                success=True, 
                method='degradation',
                note="已简化需求以完成任务"
            )
        except:
            pass
    
    # Level 4: 失败
    return RecoveryResult(success=False, method='manual_intervention')
```
