### 1.2 缓存数据分类

#### 1.2.1 公司仪表盘缓存（高频读取）

| 缓存 Key | 数据内容 | TTL | 失效策略 |
|---------|---------|-----|---------|
| `OPC:dashboard:{company_id}` | 公司仪表盘聚合数据 | 60s | 主动失效 + 定时刷新 |
| `OPC:dashboard:{company_id}:stats` | 实时统计数字 | 30s | 任务/财务事件触发失效 |
| `OPC:dashboard:{company_id}:activities` | 最近活动列表 | 120s | 任何状态变更触发失效 |

```python
# 仪表盘缓存实现示例
import json
import redis.asyncio as redis
from datetime import timedelta

class DashboardCache:
    PREFIX = "OPC:dashboard"
    STATS_TTL = 30
    ACTIVITIES_TTL = 120
    FULL_TTL = 60
    
    def __init__(self, redis_client: redis.Redis):
        self.redis = redis_client
    
    async def get(self, company_id: str) -> dict | None:
        """获取仪表盘缓存"""
        key = f"{self.PREFIX}:{company_id}"
        data = await self.redis.get(key)
        if data:
            return json.loads(data)
        return None
    
    async def set(self, company_id: str, data: dict):
        """设置仪表盘缓存"""
        key = f"{self.PREFIX}:{company_id}"
        await self.redis.setex(key, self.FULL_TTL, json.dumps(data))
    
    async def invalidate(self, company_id: str):
        """失效仪表盘缓存（事件驱动）"""
        key = f"{self.PREFIX}:{company_id}"
        await self.redis.delete(key)
        # 同时失效子缓存
        await self.redis.delete(f"{key}:stats")
        await self.redis.delete(f"{key}:activities")
    
    async def refresh_stats(self, company_id: str, stats: dict):
        """增量更新统计缓存"""
        key = f"{self.PREFIX}:{company_id}:stats"
        await self.redis.setex(key, self.STATS_TTL, json.dumps(stats))

# 事件订阅：当任务/财务事件发生时自动失效缓存
async def on_task_event(company_id: str, event_type: str):
    cache = DashboardCache(redis_client)
    await cache.invalidate(company_id)
```

#### 1.2.4 任务列表缓存

| 缓存 Key | 数据内容 | TTL | 失效策略 |
|---------|---------|-----|---------|
| `OPC:company:{company_id}:tasks:{filter_hash}` | 过滤后的任务列表 | 60s | 任务状态变更时按 filter 选择性失效 |
| `OPC:task:{task_id}` | 任务详情 | 120s | 任务更新时失效 |
| `OPC:task:{task_id}:logs` | 任务日志（最近50条） | 30s | 新日志时追加而非失效 |

```python
import hashlib

class TaskCache:
    PREFIX = "OPC:task"
    LIST_TTL = 60
    DETAIL_TTL = 120
    LOG_TTL = 30
    
    def _filter_hash(self, **filters) -> str:
        """生成过滤条件哈希"""
        filter_str = json.dumps(filters, sort_keys=True)
        return hashlib.md5(filter_str.encode()).hexdigest()[:12]
    
    async def get_list(self, company_id: str, **filters) -> list | None:
        fh = self._filter_hash(**filters)
        key = f"OPC:company:{company_id}:tasks:{fh}"
        data = await self.redis.get(key)
        return json.loads(data) if data else None
    
    async def set_list(self, company_id: str, tasks: list, **filters):
        fh = self._filter_hash(**filters)
        key = f"OPC:company:{company_id}:tasks:{fh}"
        await self.redis.setex(key, self.LIST_TTL, json.dumps(tasks))
    
    async def invalidate_company_task_lists(self, company_id: str):
        """使用 Redis Scan 清除该公司的所有任务列表缓存"""
        pattern = f"OPC:company:{company_id}:tasks:*"
        cursor = 0
        while True:
            cursor, keys = await self.redis.scan(cursor, match=pattern, count=100)
            if keys:
                await self.redis.delete(*keys)
            if cursor == 0:
                break
    
    async def append_log(self, task_id: str, log_entry: dict):
        """任务日志使用 List 结构追加"""
        key = f"{self.PREFIX}:{task_id}:logs"
        await self.redis.lpush(key, json.dumps(log_entry))
        await self.redis.ltrim(key, 0, 49)  # 只保留最近50条
        await self.redis.expire(key, self.LOG_TTL)
```

### 1.3 缓存穿透、击穿、雪崩防护

```python
from functools import wraps
import asyncio

class CacheProtection:
    """缓存防护策略"""
    
    def __init__(self, redis_client: redis.Redis):
        self.redis = redis_client
    
    async def get_with_null_protection(self, key: str, loader, ttl: int = 300) -> any:
        """防缓存穿透：空值也缓存"""
        data = await self.redis.get(key)
        if data is not None:
            # 检查是否是空值标记
            if data == b"__NULL__":
                return None
            return json.loads(data)
        
        # 回源
        result = await loader()
        if result is None:
            await self.redis.setex(key, min(ttl, 60), "__NULL__")  # 空值短缓存
        else:
            await self.redis.setex(key, ttl, json.dumps(result))
        return result
    
    async def get_with_hotkey_protection(self, key: str, loader, ttl: int = 300) -> any:
        """防缓存击穿：互斥锁 + 异步回源"""
        data = await self.redis.get(key)
        if data is not None:
            return json.loads(data)
        
        # 尝试获取互斥锁
        lock_key = f"{key}:lock"
        lock_acquired = await self.redis.set(lock_key, "1", nx=True, ex=10)
        
        if lock_acquired:
            try:
                result = await loader()
                if result is not None:
                    await self.redis.setex(key, ttl, json.dumps(result))
                return result
            finally:
                await self.redis.delete(lock_key)
        else:
            # 其他请求等待后重试
            for _ in range(5):
                await asyncio.sleep(0.1)
                data = await self.redis.get(key)
                if data is not None:
                    return json.loads(data)
            return await loader()
    
    async def mset_with_jitter(self, keys_ttls_values: list):
        """防缓存雪崩：随机 TTL"""
        import random
        pipe = self.redis.pipeline()
        for key, ttl, value in keys_ttls_values:
            jitter = random.randint(-ttl // 10, ttl // 10)  # ±10% 抖动
            pipe.setex(key, max(1, ttl + jitter), value)
        await pipe.execute()
```

```python
async def warmup_cache(redis_client: redis.Redis, db_session):
    """服务启动时预热关键缓存"""
    cache = {
        "company": CompanyCache(redis_client),
        "employee": EmployeeCache(redis_client),
        "config": ConfigCache(redis_client)

    # 1. 预热活跃公司列表（最近7天有活动的）
    active_companies = await db_session.execute(

          AND updated_at > NOW() - INTERVAL '7 days'
        LIMIT 500
        """
    )
    for company in active_companies:
        await cache["company"].set(company)
    
    # 2. 预热 MCP 工具配置
    await cache["config"].get_mcp_tools()
```

```sql
-- ❌ 低效查询：多次聚合
SELECT * FROM one_companies WHERE id = ?;

SELECT COUNT(*) FROM ai_tasks WHERE company_id = ? AND status = 'review';
SELECT SUM(amount) FROM finance_records WHERE company_id = ? AND record_type = 'revenue';

-- ✅ 优化查询：单次 CTE 聚合
WITH task_stats AS (
    SELECT 
        company_id,
        COUNT(*) FILTER (WHERE status = 'in_progress') as active_tasks,
        COUNT(*) FILTER (WHERE status = 'review') as pending_reviews,
        COUNT(*) FILTER (WHERE status = 'completed' AND completed_at > NOW() - INTERVAL '7 days') as weekly_completed
    FROM ai_tasks
    WHERE company_id = :company_id
    GROUP BY company_id
),
finance_stats AS (
    SELECT 
        company_id,
        SUM(amount_cny) FILTER (WHERE record_type = 'revenue' AND recorded_at > NOW() - INTERVAL '7 days') as weekly_revenue,
        SUM(amount_cny) FILTER (WHERE record_type = 'cost' AND recorded_at > NOW() - INTERVAL '7 days') as weekly_cost
    FROM finance_records
    WHERE company_id = :company_id
    GROUP BY company_id
)
SELECT 
    c.*,
    COALESCE(t.active_tasks, 0) as active_tasks,
    COALESCE(t.pending_reviews, 0) as pending_reviews,
    COALESCE(t.weekly_completed, 0) as weekly_completed,
    COALESCE(f.weekly_revenue, 0) as weekly_revenue,
    COALESCE(f.weekly_cost, 0) as weekly_cost
FROM one_companies c
LEFT JOIN task_stats t ON c.id = t.company_id
LEFT JOIN finance_stats f ON c.id = f.company_id
WHERE c.id = :company_id;
```

### 2.3 分页策略

```python
from typing import TypeVar, Generic
from pydantic import BaseModel

T = TypeVar("T")

class PaginatedResponse(BaseModel, Generic[T]):
    items: list[T]
    meta: dict = {
        "page": 1,
        "per_page": 20,
        "total": 0,
        "total_pages": 0,
        "has_next": False,
        "has_prev": False,
        "cursor": None  # 游标分页支持
    }

class Pagination:
    """统一分页工具类"""
    
    MAX_PER_PAGE = 100
    DEFAULT_PER_PAGE = 20
    
    @staticmethod
    def offset_limit(page: int, per_page: int) -> tuple[int, int]:
        per_page = min(per_page, Pagination.MAX_PER_PAGE)
        offset = (max(page, 1) - 1) * per_page
        return offset, per_page
    
    @staticmethod
    def build_meta(page: int, per_page: int, total: int) -> dict:
        total_pages = (total + per_page - 1) // per_page
        return {
            "page": page,
            "per_page": per_page,
            "total": total,
            "total_pages": total_pages,
            "has_next": page < total_pages,
            "has_prev": page > 1
        }
    
    @staticmethod
    def cursor_paginated(query, cursor_field: str, cursor_value: any = None, limit: int = 20):
        """游标分页（适用于大数据量场景）"""

            query = query.filter(getattr(query.column_descriptions[0]['entity'], cursor_field) > cursor_value)
        return query.order_by(cursor_field).limit(limit + 1).all()
```

```python
import time
import logging
from functools import wraps
from contextlib import asynccontextmanager

logger = logging.getLogger("OPC.slow_query")

class QueryProfiler:
    """查询性能分析器"""
    
    SLOW_QUERY_THRESHOLD_MS = 200
    
    def __init__(self):
        self.stats = {
            "total_queries": 0,
            "slow_queries": 0,
            "total_time_ms": 0,
            "query_histogram": {}
        }
    
    @asynccontextmanager
    async def profile(self, query_name: str, sql: str = None):
        start = time.perf_counter()
        try:
            yield
        finally:
            elapsed_ms = (time.perf_counter() - start) * 1000
            self.stats["total_queries"] += 1
            self.stats["total_time_ms"] += elapsed_ms
            
            # 按查询名记录直方图
            bucket = self._get_bucket(elapsed_ms)
            self.stats["query_histogram"].setdefault(query_name, {}).setdefault(bucket, 0)
            self.stats["query_histogram"][query_name][bucket] += 1
            
            if elapsed_ms > self.SLOW_QUERY_THRESHOLD_MS:
                self.stats["slow_queries"] += 1
                logger.warning(
                    f"Slow query detected: {query_name}",
                    extra={
                        "query_name": query_name,
                        "elapsed_ms": round(elapsed_ms, 2),
                        "sql": sql[:500] if sql else None,
                        "threshold_ms": self.SLOW_QUERY_THRESHOLD_MS
                    }
                )
    
    def _get_bucket(self, ms: float) -> str:
        if ms < 10: return "<10ms"
        if ms < 50: return "10-50ms"
        if ms < 100: return "50-100ms"
        if ms < 200: return "100-200ms"
        if ms < 500: return "200-500ms"
        if ms < 1000: return "500ms-1s"
        return ">1s"
    
    def get_report(self) -> dict:
        total = self.stats["total_queries"]
        return {
            **self.stats,
            "avg_time_ms": round(self.stats["total_time_ms"] / total, 2) if total else 0,
            "slow_query_rate": round(self.stats["slow_queries"] / total * 100, 2) if total else 0
        }
```

```python
from celery import Celery

import pyarrow.parquet as pq

app = Celery('OPC_archive')

@app.task
def archive_mcp_logs(months_ago: int = 3):
    """归档 MCP 调用日志"""
    cutoff_date = datetime.now() - timedelta(days=30 * months_ago)
    
    # 查询待归档数据
    records = db.query(MCPCallLog).filter(
        MCPCallLog.created_at < cutoff_date
    ).all()
    
    if not records:
        return {"archived": 0}
    
    # 转换为 Parquet
    df = pd.DataFrame([r.__dict__ for r in records])
    table = pa.Table.from_pandas(df)
    
    # 写入 S3
    s3 = boto3.client('s3')
    year_month = cutoff_date.strftime("%Y_%m")
    buffer = io.BytesIO()
    pq.write_table(table, buffer, compression='zstd')
    buffer.seek(0)
    
    s3.put_object(
        Bucket='OPC-archive',
        Key=f'mcp_logs/{year_month}.parquet',
        Body=buffer.getvalue()
    )
    
    # 删除已归档数据
    db.query(MCPCallLog).filter(
        MCPCallLog.created_at < cutoff_date
    ).delete(synchronize_session=False)
    db.commit()
    
    return {"archived": len(records), "s3_key": f'mcp_logs/{year_month}.parquet'}

@app.task
def cleanup_old_deliverables(versions_to_keep: int = 3):
    """清理交付物历史版本，只保留最近 N 个版本"""
    from sqlalchemy import func
    
    # 找出需要删除的旧版本
    subquery = db.query(
        Deliverable.task_id,
        func.row_number().over(
            partition_by=Deliverable.task_id,
            order_by=Deliverable.version.desc()
        ).label("rn")
    ).subquery()
    
    old_versions = db.query(Deliverable).join(
        subquery,
        Deliverable.task_id == subquery.c.task_id
    ).filter(subquery.c.rn > versions_to_keep).all()
    
    deleted_count = 0

                Bucket='OPC-deliverables',
                Key=deliverable.storage_path
            )
        # 删除数据库记录
        db.delete(deliverable)
        deleted_count += 1
    
    db.commit()
    return {"deleted_old_versions": deleted_count}
```

### 4.5 冷数据查询接口

```python
class ArchiveQueryService:
    """归档数据查询服务"""
    
    def __init__(self, db_session, s3_client):
        self.db = db_session
        self.s3 = s3_client
    
    async def query_mcp_logs(self, company_id: str, start_date: date, end_date: date):
        """查询 MCP 日志（支持冷热数据合并）"""
        # 1. 查询热数据（PostgreSQL）
        hot_logs = await self.db.query(MCPCallLog).filter(
            MCPCallLog.company_id == company_id,
            MCPCallLog.created_at >= start_date,
            MCPCallLog.created_at <= end_date
        ).all()
        
        # 2. 检查是否需要查询冷数据（S3）
        cold_months = self._get_cold_months(start_date, end_date)
        cold_logs = []
        
        for month in cold_months:
            s3_key = f"mcp_logs/{month.strftime('%Y_%m')}.parquet"
            try:
                obj = self.s3.get_object(Bucket='OPC-archive', Key=s3_key)
                df = pq.read_table(io.BytesIO(obj['Body'].read())).to_pandas()
                month_logs = df[df['company_id'] == company_id].to_dict('records')
                cold_logs.extend(month_logs)
            except self.s3.exceptions.NoSuchKey:
                continue
        
        # 3. 合并并按时间排序
        all_logs = hot_logs + cold_logs
        all_logs.sort(key=lambda x: x['created_at'], reverse=True)
        return all_logs
    
    def _get_cold_months(self, start_date: date, end_date: date) -> list[date]:
        """获取需要查询 S3 的月份列表（6个月前的数据）"""
        cutoff = datetime.now() - timedelta(days=180)
        months = []
        current = start_date
        while current <= end_date:
            if current < cutoff:
                months.append(current)
            current = (current.replace(day=1) + timedelta(days=32)).replace(day=1)
        return months
```
