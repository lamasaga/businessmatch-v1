## 一、总体架构

### 1.1 通信模型

```
┌─────────────────────────────────────────────────────────────────────────────┐
│                         OPC 实时通信架构                                     │
├─────────────────────────────────────────────────────────────────────────────┤
│                                                                             │
│   ┌──────────────┐      WebSocket       ┌──────────────────────────────┐   │
│   │  React Frontend │ ◄────────────────► │  FastAPI + WebSocket Manager │   │
│   │              │      (wss://...)     │                              │   │
│   │  useOPCWebSocket                      │  ┌────────────────────────┐ │   │
│   │  Hook        │                        │  │ Redis Pub/Sub          │ │   │
│   └──────────────┘                        │  │ (多实例消息广播)        │ │   │
│          │                                │  └────────────────────────┘ │   │
│          │                                │                              │   │
│          │    ┌─────────────────────┐    │  ┌────────────────────────┐ │   │
│          └───►│ Zustand OPCStore    │◄───┘  │ Celery Worker          │ │   │
│               │ (状态同步中心)        │       │ (AI任务执行引擎)        │ │   │
│               └─────────────────────┘       └────────────────────────┘ │   │
│                                                                             │
└─────────────────────────────────────────────────────────────────────────────┘
```

### 1.2 连接生命周期

```
连接建立流程:

1. 用户登录成功 → 获取 access_token
2. 进入 /opc/* 路由 → useOPCWebSocket 初始化
3. WebSocket 握手请求携带 `Authorization: Bearer <token>`
4. 后端验证 token → 建立连接 → 绑定 user_id 到连接
5. 前端发送 `subscribe_company` 消息 → 订阅该公司所有频道
6. 连接就绪，开始接收推送

心跳机制:
- 前端每 30 秒发送 `ping`
- 后端收到后回复 `pong`
- 超过 60 秒未收到 `pong` → 触发重连
```

## 二、WebSocket 事件定义

### 2.1 事件命名规范

采用 `{domain}:{action}` 的命名方式：
- `domain`: `employee` / `task` / `pipeline` / `finance` / `bmc` / `system`
- `action`: `update` / `create` / `delete` / `progress` / `notify` / `complete`

## 三、消息格式规范

### 3.1 基础消息结构

所有 WebSocket 消息采用 JSON 格式，统一包装结构：

```typescript
// 服务端 → 客户端 消息
interface WsServerMessage<T = unknown> {
  event: string;           // 事件名，如 "employee:progress"
  timestamp: string;       // ISO 8601 时间戳
  company_id: string;      // 公司ID，用于客户端过滤
  payload: T;              // 事件具体数据
  meta?: {
    seq: number;           // 消息序列号，用于排序和去重
    retry?: boolean;       // 是否为重发消息
  };
}

// 客户端 → 服务端 消息
interface WsClientMessage<T = unknown> {
  action: string;          // 动作名，如 "subscribe", "ping"
  payload?: T;             // 动作参数
  request_id?: string;     // 请求ID，用于匹配响应
}
```

## 四、断线重连策略

### 4.1 重连状态机

```
┌──────────┐    连接失败/异常断开     ┌──────────┐
│  CONNECTED│ ──────────────────────► │ DISCONNECTED│
└──────────┘                          └─────┬────┘
      ▲                                     │
      │ 重连成功                            │ 触发重连
      │                                     ▼
      │                               ┌──────────┐
      │                               │ RECONNECTING│
      │                               └─────┬────┘
      │                                     │
      │         达到最大重试次数             │ 指数退避等待
      │        ──────────────────►          │
      │        ┌──────────┐                 │
      └────────┤  FAILED   │◄────────────────┘
               └──────────┘
```

### 4.2 重连参数配置

```typescript
// src/config/websocket.ts
export const WS_CONFIG = {
  // 连接配置
  url: import.meta.env.VITE_WS_URL || 'wss://api.bizsim.edu/ws/opc',
  
  // 心跳配置
  heartbeatInterval: 30000,      // 30秒发送一次ping
  heartbeatTimeout: 60000,       // 60秒未收到pong则认为断开
  
  // 重连配置
  reconnect: {
    maxRetries: 10,              // 最大重试次数
    initialDelay: 1000,          // 首次重连延迟1秒
    maxDelay: 30000,             // 最大延迟30秒
    backoffMultiplier: 2,        // 指数退避乘数
  },
  
  // 消息队列配置

    maxSize: 100,                // 离线消息最大缓存数

  },

```

### 4.3 指数退避重连算法

```typescript
// src/hooks/useOPCWebSocket.ts
function calculateReconnectDelay(attempt: number): number {
  const { initialDelay, maxDelay, backoffMultiplier } = WS_CONFIG.reconnect;
  const delay = initialDelay * Math.pow(backoffMultiplier, attempt);
  return Math.min(delay, maxDelay);
  // attempt=0: 1000ms
  // attempt=1: 2000ms
  // attempt=2: 4000ms
  // attempt=3: 8000ms
  // attempt=4: 16000ms
  // attempt=5+: 30000ms (封顶)
}
```

### 4.4 完整 WebSocket Hook 实现

```typescript
// src/hooks/useOPCWebSocket.ts
import { useEffect, useRef, useCallback, useState } from 'react';
import { useAuthStore } from '../stores/authStore';
import { useOPCStore } from '../stores/opcStore';
import { WS_CONFIG } from '../config/websocket';

export type WsConnectionState = 'connecting' | 'connected' | 'disconnected' | 'reconnecting' | 'failed';

interface useOPCWebSocketReturn {
  connectionState: WsConnectionState;
  lastMessage: WsServerMessage | null;
  sendMessage: (action: string, payload?: unknown) => void;
  reconnect: () => void;
}

export function useOPCWebSocket(): useOPCWebSocketReturn {
  const { isAuthenticated, user } = useAuthStore();
  const { company } = useOPCStore();
  
  const wsRef = useRef<WebSocket | null>(null);
  const reconnectAttemptRef = useRef(0);
  const reconnectTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
  const heartbeatTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);
  const heartbeatTimeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null);
  const messageQueueRef = useRef<WsClientMessage[]>([]);
  const seqRef = useRef(0);                    // 已接收的最大序列号
  const pendingMessagesRef = useRef<Map<number, WsServerMessage>>(new Map()); // 乱序缓存
  
  const [connectionState, setConnectionState] = useState<WsConnectionState>('disconnected');
  const [lastMessage, setLastMessage] = useState<WsServerMessage | null>(null);

  // 清理心跳定时器
  const clearHeartbeat = useCallback(() => {
    if (heartbeatTimerRef.current) {
      clearInterval(heartbeatTimerRef.current);
      heartbeatTimerRef.current = null;
    }
    if (heartbeatTimeoutRef.current) {
      clearTimeout(heartbeatTimeoutRef.current);
      heartbeatTimeoutRef.current = null;
    }
  }, []);

  // 启动心跳
  const startHeartbeat = useCallback(() => {
    clearHeartbeat();
    
    heartbeatTimerRef.current = setInterval(() => {
      if (wsRef.current?.readyState === WebSocket.OPEN) {
        wsRef.current.send(JSON.stringify({
          action: 'ping',
          payload: { client_time: new Date().toISOString() }
        }));
        
        // 设置pong超时检测
        heartbeatTimeoutRef.current = setTimeout(() => {
          console.warn('[WS] 心跳超时，关闭连接触发重连');
          wsRef.current?.close();
        }, WS_CONFIG.heartbeatTimeout);
      }
    }, WS_CONFIG.heartbeatInterval);
  }, [clearHeartbeat]);

  // 处理接收到的消息
  const handleMessage = useCallback((event: MessageEvent) => {
    try {
      const message: WsServerMessage = JSON.parse(event.data);
      
      // 处理pong
      if (message.event === 'pong') {
        if (heartbeatTimeoutRef.current) {
          clearTimeout(heartbeatTimeoutRef.current);
          heartbeatTimeoutRef.current = null;
        }
        return;
      }
      
      // 序列号检查（确保消息按序处理）
      const msgSeq = message.meta?.seq;
      if (msgSeq !== undefined) {
        if (msgSeq <= seqRef.current) {
          // 重复或乱序消息，忽略
          return;
        }
        if (msgSeq > seqRef.current + 1) {
          // 消息缺失，缓存等待
          pendingMessagesRef.current.set(msgSeq, message);
          return;
        }
        
        // 顺序正确，处理并检查缓存
        seqRef.current = msgSeq;
        setLastMessage(message);
        
        // 处理缓存中的连续消息
        let nextSeq = msgSeq + 1;
        while (pendingMessagesRef.current.has(nextSeq)) {
          const pending = pendingMessagesRef.current.get(nextSeq)!;
          pendingMessagesRef.current.delete(nextSeq);
          setLastMessage(pending);
          seqRef.current = nextSeq;
          nextSeq++;
        }
      } else {
        // 无序列号的消息直接处理
        setLastMessage(message);
      }
    } catch (err) {
      console.error('[WS] 消息解析失败:', err);
    }
  }, []);

  // 建立连接
  const connect = useCallback(() => {
    if (!isAuthenticated || !user || !company) return;
    
    const token = localStorage.getItem('accessToken');
    if (!token) return;
    
    setConnectionState(prev => prev === 'disconnected' ? 'connecting' : 'reconnecting');
    
    const wsUrl = `${WS_CONFIG.url}?token=${encodeURIComponent(token)}`;
    const ws = new WebSocket(wsUrl);
    wsRef.current = ws;
    
    ws.onopen = () => {
      console.log('[WS] 连接成功');
      reconnectAttemptRef.current = 0;
      setConnectionState('connected');
      
      // 订阅公司频道
      ws.send(JSON.stringify({
        action: 'subscribe',
        payload: { channels: [`company:${company.id}`] }
      }));
      
      // 发送离线期间积压的消息
      while (messageQueueRef.current.length > 0) {
        const msg = messageQueueRef.current.shift();
        if (msg) ws.send(JSON.stringify(msg));
      }
      
      // 启动心跳
      startHeartbeat();
      
      // 触发全量数据刷新（补偿离线期间的状态变更）
      refreshStateAfterReconnect();
    };
    
    ws.onmessage = handleMessage;
    
    ws.onclose = (event) => {
      console.log('[WS] 连接关闭:', event.code, event.reason);
      clearHeartbeat();
      wsRef.current = null;
      
      // 非主动关闭则触发重连
      if (event.code !== 1000 && event.code !== 1001) {
        attemptReconnect();
      } else {
        setConnectionState('disconnected');
      }
    };
    
    ws.onerror = (error) => {
      console.error('[WS] 连接错误:', error);
      ws.close();
    };
  }, [isAuthenticated, user, company, handleMessage, startHeartbeat, clearHeartbeat]);

  // 重连逻辑
  const attemptReconnect = useCallback(() => {
    const { maxRetries } = WS_CONFIG.reconnect;
    
    if (reconnectAttemptRef.current >= maxRetries) {
      setConnectionState('failed');
      return;
    }
    
    const delay = calculateReconnectDelay(reconnectAttemptRef.current);
    reconnectAttemptRef.current++;
    
    setConnectionState('reconnecting');
    
    reconnectTimerRef.current = setTimeout(() => {
      connect();
    }, delay);
  }, [connect]);

  // 发送消息（自动排队）
  const sendMessage = useCallback((action: string, payload?: unknown) => {
    const message: WsClientMessage = { action, payload };
    
    if (wsRef.current?.readyState === WebSocket.OPEN) {
      wsRef.current.send(JSON.stringify(message));
    } else {
      // 离线排队
      if (messageQueueRef.current.length < WS_CONFIG.messageQueue.maxSize) {
        messageQueueRef.current.push(message);
      }
    }
  }, []);

  // 手动重连
  const reconnect = useCallback(() => {
    reconnectAttemptRef.current = 0;
    if (reconnectTimerRef.current) {
      clearTimeout(reconnectTimerRef.current);
    }
    wsRef.current?.close(1000, 'manual_reconnect');
    connect();
  }, [connect]);

  // 断线后刷新状态（补偿机制）
  const refreshStateAfterReconnect = useCallback(async () => {
    const { checkOhbAccess, setEmployees, setTasks, setBmcCanvas, setFinanceOverview } = useOPCStore.getState();
    
    // 重新拉取核心数据
    await checkOhbAccess();
    
    // 并行拉取各模块最新数据
    const [employeesRes, tasksRes, bmcRes, financeRes] = await Promise.allSettled([
      api.get('/api/v1/opc/employees'),
      api.get('/api/v1/opc/tasks'),
      api.get('/api/v1/opc/bmc/current'),
      api.get('/api/v1/opc/finance/overview'),
    ]);
    
    if (employeesRes.status === 'fulfilled') setEmployees(employeesRes.value.data.data);
    if (tasksRes.status === 'fulfilled') setTasks(tasksRes.value.data.data);
    if (bmcRes.status === 'fulfilled') setBmcCanvas(bmcRes.value.data.data);
    if (financeRes.status === 'fulfilled') setFinanceOverview(financeRes.value.data.data);
  }, []);

  // 初始化和清理
  useEffect(() => {
    if (isAuthenticated && company) {
      connect();
    }
    
    return () => {
      clearHeartbeat();
      if (reconnectTimerRef.current) clearTimeout(reconnectTimerRef.current);
      if (wsRef.current) {
        wsRef.current.close(1000, 'component_unmount');
        wsRef.current = null;
      }
    };
  }, [isAuthenticated, company, connect, clearHeartbeat]);

  return {
    connectionState,
    lastMessage,
    sendMessage,
    reconnect,
  };
}
```

```typescript
        if (!bmcCanvas) return;

          message: p.message,
```

```typescript
// src/stores/notificationStore.ts
import { create } from 'zustand';

interface Notification {
  id: string;
  type: 'info' | 'success' | 'warning' | 'error';
  title: string;
  message: string;
  action?: { label: string; url: string };
  createdAt: number;
  read: boolean;
}

interface NotificationState {
  notifications: Notification[];
  unreadCount: number;
  
  addNotification: (notification: Omit<Notification, 'id' | 'createdAt' | 'read'>) => void;
  markAsRead: (id: string) => void;
  markAllAsRead: () => void;
  dismissNotification: (id: string) => void;
  clearAll: () => void;
}

export const useNotificationStore = create<NotificationState>((set, get) => ({
  notifications: [],
  unreadCount: 0,

  addNotification: (notification) => {
    const id = `notif_${Date.now()}_${Math.random().toString(36).slice(2, 9)}`;
    const newNotification: Notification = {
      ...notification,
      id,
      createdAt: Date.now(),
      read: false,
    };
    set((state) => ({
      notifications: [newNotification, ...state.notifications].slice(0, 50),
      unreadCount: state.unreadCount + 1,
    }));
  },

  markAsRead: (id) => {
    set((state) => ({
      notifications: state.notifications.map((n) =>
        n.id === id ? { ...n, read: true } : n
      ),
      unreadCount: Math.max(0, state.unreadCount - 1),
    }));
  },

  markAllAsRead: () => {
    set((state) => ({
      notifications: state.notifications.map((n) => ({ ...n, read: true })),
      unreadCount: 0,
    }));
  },

  dismissNotification: (id) => {
    set((state) => {
      const notification = state.notifications.find((n) => n.id === id);
      return {
        notifications: state.notifications.filter((n) => n.id !== id),
        unreadCount: notification && !notification.read
          ? Math.max(0, state.unreadCount - 1)
          : state.unreadCount,
      };
    });
  },

  clearAll: () => set({ notifications: [], unreadCount: 0 }),
}));

// 便捷函数
export function showNotification(
  notification: Omit<Notification, 'id' | 'createdAt' | 'read'>
) {
  useNotificationStore.getState().addNotification(notification);
}
```

```typescript
interface VersionedState<T> {
  data: T;
  version: number;          // 递增版本号
  updatedAt: string;
}

// WebSocket 消息也携带版本号
interface WsServerMessage<T = unknown> {
  event: string;
  timestamp: string;
  company_id: string;
  payload: T;
  meta?: {
    seq: number;
    version?: number;       // 数据版本号
  };
}

// 更新时检查版本
function mergeState<T>(
  current: VersionedState<T>,
  incoming: VersionedState<T>
): VersionedState<T> {
  if (incoming.version < current.version) {
    return current;         // 忽略旧版本数据
  }
  return incoming;
}
```

## 八、安全考虑

### 8.1 认证与鉴权

- WebSocket 握手时通过 URL Query 传递 `token`
- 服务端验证 token 有效性后才建立连接
- 每个消息携带 `company_id`，服务端验证用户是否有权限访问该公司数据
- 敏感操作（如解雇员工、阶段推进）需要二次校验用户身份
