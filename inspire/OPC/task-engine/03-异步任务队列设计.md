```python
# celeryconfig.py

from celery import Celery
from kombu import Queue, Exchange

app = Celery('OPC')

# 队列定义
app.conf.task_queues = (
    Queue('OPCg.high', Exchange('OPCg.high'), routing_key='OPCg.high'),
    Queue('OPCg.default', Exchange('OPCg.default'), routing_key='OPCg.default'),
    Queue('OPCg.low', Exchange('OPCg.low'), routing_key='OPCg.low'),
    Queue('OPCg.scheduled', Exchange('OPCg.scheduled'), routing_key='OPCg.scheduled'),
)

# 默认队列
app.conf.task_default_queue = 'OPCg.default'
app.conf.task_default_exchange = 'OPCg.default'
app.conf.task_default_routing_key = 'OPCg.default'

# 路由规则
app.conf.task_routes = {
    'OPC.tasks.gateway.*': {'queue': 'OPCg.high'},
    'OPC.tasks.safety.*': {'queue': 'OPCg.high'},
    'OPC.tasks.notification.*': {'queue': 'OPCg.high'},
    'OPC.tasks.ai.*': {'queue': 'OPCg.default'},
    'OPC.tasks.report.*': {'queue': 'OPCg.default'},
    'OPC.tasks.archive.*': {'queue': 'OPCg.low'},
    'OPC.tasks.cleanup.*': {'queue': 'OPCg.low'},
    'OPC.tasks.scheduled.*': {'queue': 'OPCg.scheduled'},
}

# 结果后端
app.conf.result_backend = 'redis://localhost:6379/1'
app.conf.result_expires = 3600 * 24  # 结果保留24小时

# 任务序列化
```

### 3.1 AI任务执行

```python
@app.task(
    bind=True,
    queue='OPCg.default',
    max_retries=3,
    default_retry_delay=60,
    time_limit=3600,  # 1小时超时
    soft_time_limit=3000  # 50分钟软超时（提前警告）
)
def execute_ai_task(self, task_id: str):
    """执行AI员工任务"""
    try:
        task = load_task(task_id)
        
        # 更新状态
        task.status = 'in_progress'
        task.started_at = datetime.utcnow()
        db.commit()
        
        # 执行引擎
        engine = AIExecutionEngine()
        deliverable = await engine.execute(task)
        
        # 完成
        task.status = 'completed'
        task.completed_at = datetime.utcnow()
        db.commit()
        
        # 发送通知
        notify_task_completed.delay(task_id)
        
        return {'status': 'completed', 'task_id': task_id}
        
    except SoftTimeLimitExceeded:
        # 软超时：尝试优雅保存状态
        task.is_stuck = True
        db.commit()
        notify_student(task.company.student_id, "任务执行时间较长，正在处理...")
        raise self.retry(countdown=300)
        
    except TimeLimitExceeded:
        # 硬超时：标记失败
        task.status = 'failed'
        task.error = 'Execution timeout'
        db.commit()
        raise
        
    except Exception as exc:
        # 其他错误：重试
        if self.request.retries < self.max_retries:
            raise self.retry(exc=exc, countdown=60 * (self.request.retries + 1))
        else:
            task.status = 'failed'
            task.error = str(exc)
            db.commit()
            raise
```

```python
@app.task(queue='OPCg.high', time_limit=300)
def review_gateway(company_id: str, stage: str, review_id: str):
    """AI评审委员会评估阶段成果"""
    company = load_company(company_id)
    evidence = load_gateway_evidence(review_id)
    
    # 多维度评分
    scores = {}
    
    # 1. 完整性检查
    scores['completeness'] = check_completeness(evidence, stage)
    
    # 2. 质量评估
    scores['quality'] = evaluate_quality(evidence, stage)
    
    # 3. 数据真实性
    scores['authenticity'] = verify_authenticity(evidence)
    
    # 4. 学生匹配度
    scores['fit'] = assess_student_fit(company, evidence)
    
    # 综合决策
    total_score = weighted_average(scores)
    
    if total_score >= 80:
        decision = 'pass'
    elif total_score >= 60:
        decision = 'conditional'
    else:
        decision = 'fail'
    
    # 保存结果
    save_gateway_result(review_id, decision, scores)
    
    # 发送通知
    notify_gateway_result.delay(company_id, stage, decision)
    
    return {'review_id': review_id, 'decision': decision, 'score': total_score}
```

```python
    # WebSocket推送
    ws_connections = get_user_ws_connections(user_id)

        send_email_notification.delay(user_id, notification)
```

```python
@app.task(queue='OPCg.deadletter')
def handle_dead_letter(task_name: str, args: tuple, kwargs: dict, reason: str):
    """处理死信任务"""
    
    # 1. 记录到死信日志
    log_dead_letter(task_name, args, kwargs, reason)
    
    # 2. 分类处理
    if reason == 'max_retries_exceeded':
        # 通知管理员
        alert_admin(f"任务 {task_name} 重试失败", {'args': args, 'reason': reason})
        
        # 如果是AI任务，标记为失败
        if 'ai_task' in task_name:
            task_id = kwargs.get('task_id') or args[0]
            mark_task_failed(task_id, reason)
    
    elif reason == 'execution_timeout':
        # 可能需要人工介入
        create_manual_review_ticket(task_name, args, kwargs)
    
    elif reason == 'invalid_message':
        # 丢弃并告警
        alert_admin(f"无效消息: {task_name}")
    
    # 3. 可选：发送告警
    send_alert_if_critical(task_name, reason)
```

## 六、任务幂等性

### 6.1 幂等性保障

```python
def idempotent_task(func):
    """任务幂等性装饰器"""
    @wraps(func)
    def wrapper(self, *args, **kwargs):
        # 生成唯一任务键
        task_key = f"{func.__name__}:{hash_args(args, kwargs)}"
        
        # 检查是否已执行
        if redis.get(f"task:executed:{task_key}"):
            logger.info(f"任务 {task_key} 已执行，跳过")
            return {'status': 'skipped', 'reason': 'already_executed'}
        
        # 执行并记录
        result = func(self, *args, **kwargs)
        
        # 标记已执行（设置过期时间）
        redis.setex(f"task:executed:{task_key}", 86400, '1')
        
        return result
    return wrapper
```
