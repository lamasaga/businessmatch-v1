### 1.2 解锁算法

```python
class OPCUnlockEngine:
    def check_eligibility(self, career_id: str) -> UnlockResult:
        career = get_career_profile(career_id)
        
        results = []
        
        # 检查能力阈值
        avg_ability = sum(career.abilities.values()) / len(career.abilities)
        if avg_ability >= 70:
            results.append(UnlockCondition(
                type='ability_threshold',
                met=True,
                score=avg_ability
            ))
        
        # 检查课程完成
        completed_courses = get_completed_courses(career_id)
        if 'entrepreneurship_101' in completed_courses:
            results.append(UnlockCondition(
                type='course_completion',
                met=True
            ))
        
        # 检查商赛成绩
        arena_results = get_arena_results(career_id)
        for mode in ['startup_survival', 'investor_duel']:
            if mode in arena_results and arena_results[mode].rank_percentile <= 0.3:
                results.append(UnlockCondition(
                    type='arena_achievement',
                    met=True,
                    game_mode=mode
                ))
        
        # 检查徽章
        badges = get_earned_badges(career_id)
        required = {'strategic_master', 'financial_insight_lv3'}
        if required.issubset(set(badges)):
            results.append(UnlockCondition(
                type='badge_combination',
                met=True
            ))
        
        # 综合判断：满足任一条件即可
        is_eligible = any(r.met for r in results)
        
        return UnlockResult(
            eligible=is_eligible,
            conditions=results,
            recommended_path=self.get_recommended_path(career)
        )
```

## 二、认证对接

### 2.1 JWT 共享

```yaml
auth_integration:
  token_format: JWT
  shared_secret: "通过环境变量共享"
  
  token_payload:
    sub: user_id
    career_id: career_id
    roles: [student, teacher, ...]
    iat: 签发时间
    exp: 过期时间
    OPC_access: true/false  # 是否已解锁OPC
  
  validation:
    - 验证签名
    - 检查过期时间
    - 检查OPC_access标志
    - 检查用户状态（未被禁用）
```

### 2.2 跨域认证

```python
# OPC API中验证来自现有平台的Token

async def verify_platform_token(token: str) -> UserInfo:
    try:
        payload = jwt.decode(token, PLATFORM_PUBLIC_KEY, algorithms=["RS256"])
        
        # 检查OPC访问权限
        if not payload.get('OPC_access'):
            raise HTTPException(403, "OPC未解锁")
        
        return UserInfo(
            user_id=payload['sub'],
            career_id=payload['career_id'],
            roles=payload['roles']
        )
    except jwt.ExpiredSignatureError:
        raise HTTPException(401, "Token已过期")
    except jwt.InvalidTokenError:
        raise HTTPException(401, "无效的Token")
```
